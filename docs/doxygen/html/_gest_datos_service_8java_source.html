52 public long guardarTarea(Tarea tarea, List<Long> idUsuarios){ // <-- Cambiado a List<Long>
53 if (tarea == null) {
54 throw new IllegalArgumentException("La tarea no puede ser nula");
55 }
56
57 if (tarea.getFechaInicio() != null && tarea.getFechaFin() != null && tarea.getFechaInicio().isAfter(tarea.getFechaFin())) {
58 throw new IllegalArgumentException("La fecha de inicio no puede ser posterior a la fecha de fin");
59 }
60

62 tarea.setUsuarios(new ArrayList<>());
